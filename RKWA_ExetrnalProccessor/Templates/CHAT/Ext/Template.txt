        /**

        const WhatsAppChatAPI = {
            // Настройки инстанса
            instanceSettings: {
                instanceId: null,
                authToken: null,
                authType: 'jwt', // 'jwt' или 'apikey'
                apiBaseUrl: '/api/whatsapp',
                longPollingInterval: 5000,
                longPollingActive: false,
                latestTimestamp: null, // для хранения последней временной метки сообщений
                onMessageReceived: null, // колбэк для обработки новых сообщений в 1С
            },
            
            // Установка настроек инстанса
            setInstanceSettings: function(settings) {
                // Валидация обязательных полей
                if (!settings.instanceId) {
                    console.error('Не указан instanceId');
                    return false;
                }
                
                // Обновляем настройки
                this.instanceSettings.instanceId = settings.instanceId;
                
                if (settings.authToken) {
                    this.instanceSettings.authToken = settings.authToken;
                }
                
                if (settings.authType && (settings.authType === 'jwt' || settings.authType === 'apikey')) {
                    this.instanceSettings.authType = settings.authType;
                }
                
                if (settings.apiBaseUrl) {
                    this.instanceSettings.apiBaseUrl = settings.apiBaseUrl;
                }
                
                if (settings.longPollingInterval) {
                    this.instanceSettings.longPollingInterval = parseInt(settings.longPollingInterval);
                }
                
                if (typeof settings.onMessageReceived === 'function') {
                    this.instanceSettings.onMessageReceived = settings.onMessageReceived;
                }
                
                return true;
            },
            
            // Установка текущего контакта
            setCurrentContact: function(name, number) {
                // Установка имени и номера
                document.getElementById('chat-name').textContent = name;
                document.getElementById('chat-number').textContent = number;
                
                // Установка первой буквы имени в аватар
                const firstLetter = name.charAt(0).toUpperCase();
                document.getElementById('chat-avatar').textContent = firstLetter;
                
                // Очистка предыдущих сообщений
                document.getElementById('chat-messages').innerHTML = '';
                
                return true;
            },
            
            // Добавление входящего сообщения
            addIncomingMessage: function(data) {
                const messagesContainer = document.getElementById('chat-messages');
                
                // Значения по умолчанию
                const messageData = {
                    text: data.text || '',
                    time: data.time || this.getCurrentTime(),
                    type: data.type || 'text', // text, image, file
                    fileUrl: data.fileUrl || null,
                    fileName: data.fileName || null,
                    fileSize: data.fileSize || null,
                    mimeType: data.mimeType || null
                };
                
                // Создаем элемент сообщения
                const messageElement = document.createElement('div');
                messageElement.classList.add('message', 'incoming');
                
                // Обрабатываем разные типы сообщений
                if (messageData.type === 'image') {
                    // Добавляем изображение
                    const mediaElement = document.createElement('div');
                    mediaElement.classList.add('message-media');
                    
                    const imgElement = document.createElement('img');
                    imgElement.src = messageData.fileUrl;
                    imgElement.alt = 'Изображение';
                    mediaElement.appendChild(imgElement);
                    
                    messageElement.appendChild(mediaElement);
                    
                    // Если есть подпись к изображению
                    if (messageData.text) {
                        const textElement = document.createElement('div');
                        textElement.classList.add('message-text');
                        textElement.textContent = messageData.text;
                        messageElement.appendChild(textElement);
                    }
                }
                else if (messageData.type === 'file') {
                    // Добавляем файл
                    const fileElement = document.createElement('div');
                    fileElement.classList.add('message-file');
                    
                    // Иконка файла
                    const iconElement = document.createElement('div');
                    iconElement.classList.add('file-icon');
                    iconElement.innerHTML = '';
                    fileElement.appendChild(iconElement);
                    
                    // Информация о файле
                    const infoElement = document.createElement('div');
                    infoElement.classList.add('file-info');
                    
                    const nameElement = document.createElement('div');
                    nameElement.classList.add('file-name');
                    nameElement.textContent = messageData.fileName || 'Файл';
                    infoElement.appendChild(nameElement);
                    
                    if (messageData.fileSize) {
                        const sizeElement = document.createElement('div');
                        sizeElement.classList.add('file-size');
                        sizeElement.textContent = this.formatFileSize(messageData.fileSize);
                        infoElement.appendChild(sizeElement);
                    }
                    
                    fileElement.appendChild(infoElement);
                    
                    // Кнопка скачивания
                    if (messageData.fileUrl) {
                        const downloadElement = document.createElement('a');
                        downloadElement.classList.add('file-download');
                        downloadElement.href = messageData.fileUrl;
                        downloadElement.target = '_blank';
                        downloadElement.innerHTML = '⬇️';
                        downloadElement.title = 'Скачать файл';
                        fileElement.appendChild(downloadElement);
                    }
                    
                    messageElement.appendChild(fileElement);
                }
                else {
                    // Обычное текстовое сообщение
                    const textElement = document.createElement('div');
                    textElement.classList.add('message-text');
                    textElement.textContent = messageData.text;
                    messageElement.appendChild(textElement);
                }
                
                // Добавляем время сообщения
                const timeElement = document.createElement('div');
                timeElement.classList.add('message-time');
                timeElement.textContent = messageData.time;
                messageElement.appendChild(timeElement);
                
                // Добавляем сообщение в контейнер
                messagesContainer.appendChild(messageElement);
                
                // Прокручиваем к последнему сообщению
                messagesContainer.scrollTop = messagesContainer.scrollHeight;
                
                return true;
            },

            },
            
            // Получение текущего времени в формате HH:MM
            getCurrentTime: function() {
                const now = new Date();
                const hours = now.getHours().toString().padStart(2, '0');
                const minutes = now.getMinutes().toString().padStart(2, '0');
                return `${hours}:${minutes}`;
            },
            
            // Форматирование размера файла
            formatFileSize: function(bytes) {
                if (bytes < 1024) {
                    return bytes + ' байт';
                } else if (bytes < 1024 * 1024) {
                    return (bytes / 1024).toFixed(1) + ' КБ';
                } else if (bytes < 1024 * 1024 * 1024) {
                    return (bytes / (1024 * 1024)).toFixed(1) + ' МБ';
                } else {
                    return (bytes / (1024 * 1024 * 1024)).toFixed(1) + ' ГБ';
                }
            },
            
            // Очистка всех сообщений в чате
            clearMessages: function() {
                document.getElementById('chat-messages').innerHTML = '';
                return true;
            },
            
            // Добавление группы сообщений
            addMessages: function(messages) {
                if (!Array.isArray(messages)) return false;
                
                for (const message of messages) {
                    if (message.type === 'incoming') {
                        this.addIncomingMessage(message);
                    } else if (message.type === 'outgoing') {
                        this.addOutgoingMessage(message);
                    }
                }
                
                return true;
            },
            
            // Запуск longpolling для получения новых сообщений
            startLongPolling: function() {
                if (!this.instanceSettings.instanceId || !this.instanceSettings.authToken) {
                    console.error('Для longpolling необходимо настроить instanceId и authToken');
                    return false;
                }
                
                this.instanceSettings.longPollingActive = true;
                this.fetchNewMessages();
                
                console.log('LongPolling запущен для инстанса', this.instanceSettings.instanceId);
                return true;
            },
            
            // Остановка longpolling
            stopLongPolling: function() {
                this.instanceSettings.longPollingActive = false;
                console.log('LongPolling остановлен для инстанса', this.instanceSettings.instanceId);
                return true;
            },
            
            // Загрузка новых сообщений с сервера
            fetchNewMessages: function() {
                if (!this.instanceSettings.longPollingActive) return;
                
                const self = this;
                const instanceId = this.instanceSettings.instanceId;
                const baseUrl = this.instanceSettings.apiBaseUrl;
                
                // Создаем URL для запроса
                let url = `${baseUrl}/${instanceId}/events?types=message_received&limit=50`;
                
                // Добавляем timestamp последнего сообщения, если есть
                if (this.instanceSettings.latestTimestamp) {
                    url += `&since=${encodeURIComponent(this.instanceSettings.latestTimestamp)}`;
                }
                
                // Подготавливаем заголовки с авторизацией
                const headers = new Headers();
                
                if (this.instanceSettings.authType === 'jwt') {
                    headers.append('Authorization', `Bearer ${this.instanceSettings.authToken}`);
                } else if (this.instanceSettings.authType === 'apikey') {
                    headers.append('X-API-Key', this.instanceSettings.authToken);
                }
                
                // Добавляем индикатор загрузки
                const messagesContainer = document.getElementById('chat-messages');
                const loadingIndicator = document.createElement('div');
                loadingIndicator.classList.add('loading-indicator');
                loadingIndicator.textContent = 'Обновление сообщений...';
                loadingIndicator.id = 'loading-indicator';
                
                // Делаем запрос к API
                fetch(url, {
                    method: 'GET',
                    headers: headers
                })
                .then(response => {
                    if (!response.ok) {
                        throw new Error(`HTTP error! status: ${response.status}`);
                    }
                    return response.json();
                })
                .then(data => {
                    // Если индикатор загрузки существует, удаляем его
                    const existingIndicator = document.getElementById('loading-indicator');
                    if (existingIndicator) {
                        existingIndicator.remove();
                    }
                    
                    // Обрабатываем полученные сообщения
                    if (data.events && data.events.length > 0) {
                        // Обновляем timestamp последнего сообщения
                        if (data.latestTimestamp) {
                            self.instanceSettings.latestTimestamp = data.latestTimestamp;
                        }
                        
                        // Обрабатываем каждое новое сообщение
                        data.events.forEach(event => {
                            // Обрабатываем только полученные сообщения
                            if (event.type === 'message_received') {
                                // Преобразуем формат сообщения для отображения
                                const messageData = {
                                    text: event.data.body || '',
                                    time: self.formatTimestamp(event.timestamp),
                                    type: 'text'
                                };
                                
                                // Определяем тип сообщения
                                if (event.data.hasMedia) {
                                    if (event.data.mimeType && event.data.mimeType.startsWith('image/')) {
                                        messageData.type = 'image';
                                        messageData.fileUrl = event.data.mediaUrl;
                                        messageData.caption = event.data.caption;
                                    } else {
                                        messageData.type = 'file';
                                        messageData.fileUrl = event.data.mediaUrl;
                                        messageData.fileName = event.data.fileName || 'Файл';
                                        messageData.mimeType = event.data.mimeType;
                                    }
                                }
                                
                                // Добавляем сообщение в чат
                                self.addIncomingMessage(messageData);
                                
                                // Вызываем колбэк, если он настроен
                                if (typeof self.instanceSettings.onMessageReceived === 'function') {
                                    self.instanceSettings.onMessageReceived(event.data);
                                }
                            }
                        });
                    }
                    
                    // Планируем следующий запрос
                    setTimeout(() => {
                        if (self.instanceSettings.longPollingActive) {
                            self.fetchNewMessages();
                        }
                    }, self.instanceSettings.longPollingInterval);
                })
                .catch(error => {
                    console.error('Ошибка при получении сообщений:', error);
                    
                    // Если индикатор загрузки существует, удаляем его
                    const existingIndicator = document.getElementById('loading-indicator');
                    if (existingIndicator) {
                        existingIndicator.remove();
                    }
                    
                    // Планируем следующий запрос через больший интервал при ошибке
                    setTimeout(() => {
                        if (self.instanceSettings.longPollingActive) {
                            self.fetchNewMessages();
                        }
                    }, self.instanceSettings.longPollingInterval * 2);
                });
            },
            
            // Форматирование временной метки из ISO в HH:MM
            formatTimestamp: function(timestamp) {
                try {
                    const date = new Date(timestamp);
                    if (isNaN(date.getTime())) {
                        return this.getCurrentTime();
                    }
                    
                    const hours = date.getHours().toString().padStart(2, '0');
                    const minutes = date.getMinutes().toString().padStart(2, '0');
                    return `${hours}:${minutes}`;
                } catch (error) {
                    return this.getCurrentTime();
                }
            },
            
            // Определение типа файла по MIME-типу
            getFileIcon: function(mimeType) {
                if (!mimeType) return '';
                
                if (mimeType.startsWith('image/')) {
                    return '️';
                } else if (mimeType.startsWith('audio/')) {
                    return '';
                } else if (mimeType.startsWith('video/')) {
                    return '';
                } else if (mimeType.startsWith('text/')) {
                    return '';
                } else if (mimeType.includes('pdf')) {
                    return '';
                } else if (mimeType.includes('word') || mimeType.includes('document')) {
                    return '';
                } else if (mimeType.includes('excel') || mimeType.includes('sheet')) {
                    return '';
                } else if (mimeType.includes('presentation') || mimeType.includes('powerpoint')) {
                    return '';
                } else if (mimeType.includes('zip') || mimeType.includes('compressed')) {
                    return '';
                }
                
                return '';
            },
            
            // Отправка сообщения через API (для интеграции с 1C)
            sendMessage: function(phone, message, callback) {
                if (!this.instanceSettings.instanceId || !this.instanceSettings.authToken) {
                    console.error('Для отправки сообщений необходимо настроить instanceId и authToken');
                    if (callback) callback(false, 'Не настроены параметры соединения');
                    return false;
                }
                
                const instanceId = this.instanceSettings.instanceId;
                const baseUrl = this.instanceSettings.apiBaseUrl;
                
                // Подготавливаем заголовки с авторизацией
                const headers = new Headers();
                headers.append('Content-Type', 'application/json');
                
                if (this.instanceSettings.authType === 'jwt') {
                    headers.append('Authorization', `Bearer ${this.instanceSettings.authToken}`);
                } else if (this.instanceSettings.authType === 'apikey') {
                    headers.append('X-API-Key', this.instanceSettings.authToken);
                }
                
                // Отправляем запрос на API
                fetch(`${baseUrl}/${instanceId}/send`, {
                    method: 'POST',
                    headers: headers,
                    body: JSON.stringify({
                        phone: phone,
                        message: message
                    })
                })
                .then(response => {
                    if (!response.ok) {
                        throw new Error(`HTTP error! status: ${response.status}`);
                    }
                    return response.json();
                })
                .then(data => {
                    // Добавляем сообщение в чат
                    this.addOutgoingMessage({
                        text: message,
                        time: this.getCurrentTime(),
                        type: 'text'
                    });
                    
                    if (callback) callback(true, data);
                })
                .catch(error => {
                    console.error('Ошибка при отправке сообщения:', error);
                    if (callback) callback(false, error.message);
                });
                
                return true;
            },
            
            // Отправка файла через API (для интеграции с 1C)
            sendFile: function(phone, fileUrl, caption, callback) {
                if (!this.instanceSettings.instanceId || !this.instanceSettings.authToken) {
                    console.error('Для отправки файлов необходимо настроить instanceId и authToken');
                    if (callback) callback(false, 'Не настроены параметры соединения');
                    return false;
                }
                
                const instanceId = this.instanceSettings.instanceId;
                const baseUrl = this.instanceSettings.apiBaseUrl;
                
                // Подготавливаем заголовки с авторизацией
                const headers = new Headers();
                headers.append('Content-Type', 'application/json');
                
                if (this.instanceSettings.authType === 'jwt') {
                    headers.append('Authorization', `Bearer ${this.instanceSettings.authToken}`);
                } else if (this.instanceSettings.authType === 'apikey') {
                    headers.append('X-API-Key', this.instanceSettings.authToken);
                }
                
                // Отправляем запрос на API
                fetch(`${baseUrl}/${instanceId}/send-media`, {
                    method: 'POST',
                    headers: headers,
                    body: JSON.stringify({
                        phone: phone,
                        url: fileUrl,
                        caption: caption || ''
                    })
                })
                .then(response => {
                    if (!response.ok) {
                        throw new Error(`HTTP error! status: ${response.status}`);
                    }
                    return response.json();
                })
                .then(data => {
                    // Добавляем сообщение в чат
                    // Определяем тип файла по URL
                    const isImage = fileUrl.match(/\.(jpg|jpeg|png|gif|webp)$/i) !== null;
                    
                    this.addOutgoingMessage({
                        text: caption || '',
                        time: this.getCurrentTime(),
                        type: isImage ? 'image' : 'file',
                        fileUrl: fileUrl,
                        fileName: this.getFileNameFromUrl(fileUrl)
                    });
                    
                    if (callback) callback(true, data);
                })
                .catch(error => {
                    console.error('Ошибка при отправке файла:', error);
                    if (callback) callback(false, error.message);
                });
                
                return true;
            },
            
            // Получение имени файла из URL
            getFileNameFromUrl: function(url) {
                try {
                    const urlObj = new URL(url);
                    const pathname = urlObj.pathname;
                    const filename = pathname.split('/').pop();
                    return decodeURIComponent(filename);
                } catch (e) {
                    // Если не удалось обработать как URL, берем последнюю часть строки после /
                    const parts = url.split('/');
                    return parts[parts.length - 1];
                }
            },
            
            // Загрузка истории сообщений чата
            loadChatHistory: function(phone, callback) {
                if (!this.instanceSettings.instanceId || !this.instanceSettings.authToken) {
                    console.error('Для загрузки истории необходимо настроить instanceId и authToken');
                    if (callback) callback(false, 'Не настроены параметры соединения');
                    return false;
                }
                
                const instanceId = this.instanceSettings.instanceId;
                const baseUrl = this.instanceSettings.apiBaseUrl;
                
                // Подготавливаем заголовки с авторизацией
                const headers = new Headers();
                
                if (this.instanceSettings.authType === 'jwt') {
                    headers.append('Authorization', `Bearer ${this.instanceSettings.authToken}`);
                } else if (this.instanceSettings.authType === 'apikey') {
                    headers.append('X-API-Key', this.instanceSettings.authToken);
                }
                
                // Очищаем существующие сообщения
                this.clearMessages();
                
                // Добавляем индикатор загрузки
                const messagesContainer = document.getElementById('chat-messages');
                const loadingIndicator = document.createElement('div');
                loadingIndicator.classList.add('loading-indicator');
                loadingIndicator.textContent = 'Загрузка истории сообщений...';
                loadingIndicator.id = 'history-loading-indicator';
                messagesContainer.appendChild(loadingIndicator);
                
                // Отправляем запрос на API
                fetch(`${baseUrl}/${instanceId}/chats/${phone}/messages?limit=50`, {
                    method: 'GET',
                    headers: headers
                })
                .then(response => {
                    if (!response.ok) {
                        throw new Error(`HTTP error! status: ${response.status}`);
                    }
                    return response.json();
                })
                .then(data => {
                    // Удаляем индикатор загрузки
                    const indicator = document.getElementById('history-loading-indicator');
                    if (indicator) indicator.remove();
                    
                    // Очищаем существующие сообщения
                    this.clearMessages();
                    
                    if (data.messages && data.messages.length > 0) {
                        // Сортируем сообщения по дате (от старых к новым)
                        const sortedMessages = data.messages.sort((a, b) => {
                            return new Date(a.createdAt) - new Date(b.createdAt);
                        });
                        
                        // Добавляем сообщения в чат
                        sortedMessages.forEach(message => {
                            const messageData = {
                                text: message.content || '',
                                time: this.formatTimestamp(message.createdAt),
                                type: 'text'
                            };
                            
                            // Определяем тип сообщения
                            if (message.hasMedia) {
                                if (message.mimeType && message.mimeType.startsWith('image/')) {
                                    messageData.type = 'image';
                                    messageData.fileUrl = message.mediaUrl;
                                    messageData.caption = message.caption;
                                } else {
                                    messageData.type = 'file';
                                    messageData.fileUrl = message.mediaUrl;
                                    messageData.fileName = message.fileName || 'Файл';
                                    messageData.mimeType = message.mimeType;
                                }
                            }
                            
                            // Добавляем сообщение
                            if (message.fromMe) {
                                this.addOutgoingMessage(messageData);
                            } else {
                                this.addIncomingMessage(messageData);
                            }
                        });
                        
                        // Прокручиваем к последнему сообщению
                        const messagesContainer = document.getElementById('chat-messages');
                        messagesContainer.scrollTop = messagesContainer.scrollHeight;
                    }
                    
                    if (callback) callback(true, data);
                })
                .catch(error => {
                    console.error('Ошибка при загрузке истории сообщений:', error);
                    
                    // Удаляем индикатор загрузки
                    const indicator = document.getElementById('history-loading-indicator');
                    if (indicator) indicator.remove();
                    
                    // Добавляем сообщение об ошибке
                    const messagesContainer = document.getElementById('chat-messages');
                    const errorIndicator = document.createElement('div');
                    errorIndicator.classList.add('loading-indicator');
                    errorIndicator.style.color = '#c0392b';
                    errorIndicator.textContent = 'Ошибка загрузки истории сообщений';
                    messagesContainer.appendChild(errorIndicator);
                    
                    setTimeout(() => {
                        errorIndicator.remove();
                    }, 3000);
                    
                    if (callback) callback(false, error.message);
                });
                
                return true;
            }
        };
