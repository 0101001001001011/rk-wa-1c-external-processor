    <script>
        // Функция для отображения ошибки
        function showError(message) {
            var errorEl = document.getElementById('error-message');
            errorEl.textContent = message;
            errorEl.style.display = 'block';
            console.error("Error:", message);
        }
        
        // Функция для очистки ошибки
        function clearError() {
            var errorEl = document.getElementById('error-message');
            errorEl.textContent = '';
            errorEl.style.display = 'none';
        }
        
        // Глобальный метод для вызова из внешней системы
        window.displayQRCode = function(qrString, status) {
            clearError();
            console.log("displayQRCode called with status:", status);
            
            // Скрываем все контейнеры
            document.getElementById('qrcode-container').style.display = 'none';
            document.getElementById('auth-message').style.display = 'none';
            document.getElementById('loading-message').style.display = 'none';
            
            // Проверяем статус
            if (status === 'isauth' || status === 'connected') {
                // Подключен, показываем сообщение об успешной авторизации
                document.getElementById('auth-message').style.display = 'block';
                return true;
            }
            
            if (status === 'loading') {
                // Загрузка, показываем сообщение о загрузке
                document.getElementById('loading-message').style.display = 'block';
            } else {
                // Используем статус как строку QR-кода
                var qrContainer = document.getElementById('qrcode');
                qrContainer.innerHTML = '';
                
                try {
                    // Пробуем использовать статус как строку QR-кода
                    var img = document.createElement('img');
                    img.src = 'https://api.qrserver.com/v1/create-qr-code/?size=300x300&data=' + encodeURIComponent(status);
                    qrContainer.appendChild(img);
                    
                    document.getElementById('qrcode-container').style.display = 'flex';
                    console.log("Created QR code using status as QR string, length:", status.length);
                } catch (error) {
                    showError('Ошибка генерации QR-кода: ' + error.message);
                    document.getElementById('loading-message').style.display = 'block';
                }
            }
            
            return true;
        };
        
        // Инициализация при загрузке страницы
        window.onload = function() {
            console.log("Page loaded");
            document.getElementById('loading-message').style.display = 'block';
        };
    </script>
