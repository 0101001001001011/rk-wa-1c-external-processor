# Методы интеграции с WhatsApp API в 1С

## Пример отправки текстового сообщения

```bsl
// Процедура отправки сообщения клиенту
//
// Параметры:
//  НомерТелефона - Строка - Номер телефона получателя
//  ТекстСообщения - Строка - Текст сообщения
//  ИдИнстанса - Строка - Идентификатор WhatsApp инстанса
//
Процедура ОтправитьСообщениеКлиенту(НомерТелефона, ТекстСообщения, ИдИнстанса)
    // Создаем параметры подключения
    ПараметрыСУ = РегистрыСведений.НастройкиWhatsApp.ПолучитьПараметрыПодключения(ИдИнстанса);
    
    Если ПараметрыСУ = Неопределено Тогда
        ВызватьИсключение "Не удалось получить параметры подключения для инстанса " + ИдИнстанса;
    КонецЕсли;
    
    // Проверяем корректность номера телефона
    НормализованныйНомер = Лев(СтрЗаменить(НомерТелефона, "+", ""), 10);
    
    Попытка
        // Вызываем метод отправки сообщения из внешней обработки
        Результат = ВнешняяОбработка.WhatsAppAPI.ОтправитьСообщение(
            НормализованныйНомер, 
            ТекстСообщения, 
            ПараметрыСУ
        );
        
        // Проверяем результат
        Если Результат.Успех Тогда
            ЖурналРегистрации.Информация(
                "Сообщение отправлено", 
                "WhatsApp", 
                , 
                , 
                "Номер: " + НомерТелефона + ", Текст: " + Лев(ТекстСообщения, 50)
            );
        Иначе
            ВызватьИсключение "Ошибка отправки сообщения: " + Результат.Сообщение;
        КонецЕсли;
        
    Исключение
        ЖурналРегистрации.Ошибка(
            "Ошибка отправки сообщения WhatsApp", 
            "WhatsApp", 
            , 
            , 
            ОписаниеОшибки()
        );
        
        ВызватьИсключение;
    КонецПопытки;
КонецПроцедуры

// Пример вызова
// ОтправитьСообщениеКлиенту("79001234567", "Ваш заказ готов!", "instance_01");
```

## Пример отправки файла

```bsl
// Процедура отправки файла клиенту
//
// Параметры:
//  НомерТелефона - Строка - Номер телефона получателя
//  ПутьКФайлу - Строка - Полный путь к файлу
//  Подпись - Строка - Необязательная подпись к файлу
//  ИдИнстанса - Строка - Идентификатор WhatsApp инстанса
//
Процедура ОтправитьФайлКлиенту(НомерТелефона, ПутьКФайлу, Подпись = "", ИдИнстанса)
    // Проверяем существование файла
    Если Не ФайлСуществует(ПутьКФайлу) Тогда
        ВызватьИсключение "Файл не найден: " + ПутьКФайлу;
    КонецЕсли;
    
    // Создаем параметры подключения
    ПараметрыСУ = РегистрыСведений.НастройкиWhatsApp.ПолучитьПараметрыПодключения(ИдИнстанса);
    
    Если ПараметрыСУ = Неопределено Тогда
        ВызватьИсключение "Не удалось получить параметры подключения для инстанса " + ИдИнстанса;
    КонецЕсли;
    
    // Создаем объект двоичных данных
    ДвоичныеДанные = Новый ДвоичныеДанные(ПутьКФайлу);
    
    // Получаем имя файла
    ИмяФайла = Сред(ПутьКФайлу, СтрДлина(ПутьКФайлу) - СтрНайти(Обратить(ПутьКФайлу), "\") + 2);
    
    // Проверяем корректность номера телефона
    НормализованныйНомер = Лев(СтрЗаменить(НомерТелефона, "+", ""), 10);
    
    Попытка
        // Вызываем метод отправки файла из внешней обработки
        Результат = ВнешняяОбработка.WhatsAppAPI.ОтправитьФайл(
            НормализованныйНомер, 
            ДвоичныеДанные,
            ИмяФайла, 
            ПараметрыСУ,
            Подпись
        );
        
        // Проверяем результат
        Если Результат.Успех Тогда
            ЖурналРегистрации.Информация(
                "Файл отправлен", 
                "WhatsApp", 
                , 
                , 
                "Номер: " + НомерТелефона + ", Файл: " + ИмяФайла
            );
        Иначе
            ВызватьИсключение "Ошибка отправки файла: " + Результат.Сообщение;
        КонецЕсли;
        
    Исключение
        ЖурналРегистрации.Ошибка(
            "Ошибка отправки файла WhatsApp", 
            "WhatsApp", 
            , 
            , 
            ОписаниеОшибки()
        );
        
        ВызватьИсключение;
    КонецПопытки;
КонецПроцедуры

// Примеры вызова
// 1. Отправка файла без подписи
// ОтправитьФайлКлиенту("79001234567", "C:\Files\invoice.pdf", "", "instance_01");

// 2. Отправка файла с подписью
// ОтправитьФайлКлиенту(
//     "79001234567", 
//     "C:\Files\invoice.pdf", 
//     "Счет на оплату №123 от 15.05.2023", 
//     "instance_01"
// );
```

## Пример интеграции с документом в 1С

```bsl
// Отправка уведомления о статусе заказа в WhatsApp
Процедура ОтправитьУведомлениеОЗаказе(Заказ)
    Попытка
        // Получаем номер телефона клиента
        НомерТелефона = Заказ.Клиент.Телефон;
        
        // Формируем текст сообщения
        ТекстСообщения = 
            "Уважаемый клиент!" + Символы.ПС +
            "Статус вашего заказа №" + Заказ.Номер + " от " + 
            Формат(Заказ.Дата, "ДФ=dd.MM.yyyy") + ": " + Заказ.Статус + "." + Символы.ПС +
            "Сумма заказа: " + Формат(Заказ.Сумма, "ЧЦ=10; ЧДЦ=2") + " руб." + Символы.ПС +
            "Спасибо за покупку!";
        
        // Получаем ID WhatsApp инстанса для данного клиента
        ИдИнстанса = ПолучитьИнстансДляКлиента(Заказ.Клиент);
        
        // Отправляем сообщение
        ОтправитьСообщениеКлиенту(НомерТелефона, ТекстСообщения, ИдИнстанса);
        
        // Фиксируем факт отправки уведомления
        Заказ.ОтметкаОПочтеОтправлена = Истина;
        Заказ.Записать();
        
    Исключение
        // Обработка ошибок отправки
        ЖурналРегистрации.Ошибка(
            "Ошибка уведомления о заказе", 
            "WhatsApp", 
            ,
            Заказ.Ссылка, 
            ОписаниеОшибки()
        );
    КонецПопытки;
КонецПроцедуры
```

## Замечания по использованию

1. Методы `ОтправитьСообщение` и `ОтправитьФайл` находятся во внешней обработке `WhatsAppAPI`
2. Перед использованием необходимо настроить параметры подключения в регистре сведений
3. Рекомендуется обрабатывать возможные исключения
4. Для корректной работы требуется предварительная настройка WhatsApp API
